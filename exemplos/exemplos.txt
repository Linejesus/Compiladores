Exemplo 2 — Estrutura condicional

programa

inteiro a.
inteiro b.
inteiro maior.

escreva("Digite A").
leia(a).

escreva("Digite B").
leia(b).

se(a > b) {
maior := a.
} senao {
maior := b.
}

escreva("Maior valor").
escreva(maior).

fimprograma

Exemplo 6 — Estrutura de repetição - Para

programa

inteiro soma.

soma := 0.

para(inteiro i := 1. i <= 5. i := i + 1) {
soma := soma + i.
}

escreva("Resultado da soma").
escreva(soma).

fimprograma



Exemplo 7 — Laços encadeados

programa

para(inteiro j := 1. j <= 2. j := j + 1) {
escreva(i).

}
}

fimprograma
